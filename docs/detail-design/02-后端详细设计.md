## 四、Service 层设计

### 4.1 设计原则

- **Service = 业务编排**：组合 Repository + 外部调用 + 缓存，但不直接写 SQL
- **方法粒度**：一个方法只做一件事，长流程拆分为多个 private 方法
- **事务边界**：Service 方法 = 事务边界，使用 `async with session.begin()`
- **异常**：Service 抛 `BizException`，由全局 handler 统一处理

### 4.2 TaskService 示例

```python
from typing import Sequence
from sqlalchemy.ext.asyncio import AsyncSession
from app.repositories.task_repo import TaskRepository
from app.exceptions.biz import BizException
from app.exceptions.codes import ErrorCode
from app.schemas.task import TaskCreate, TaskUpdate, TaskDTO, TaskQuery, BatchUpdateStatusRequest
from app.schemas.common import PageResponse
from app.services.action_service import ActionService

class TaskService:
    def __init__(self, session: AsyncSession, repo: TaskRepository, action_svc: ActionService):
        self.session = session
        self.repo = repo
        self.action_svc = action_svc

    async def list_tasks(self, query: TaskQuery, user_id: int) -> PageResponse[TaskDTO]:
        # 权限：FDE 只能看自己的或所在项目的
        items, total = await self.repo.search(query, viewer_id=user_id)
        return PageResponse(
            items=[TaskDTO.model_validate(t, from_attributes=True) for t in items],
            total=total, page=query.page, size=query.size,
        )

    async def get_task(self, task_id: int, user_id: int) -> TaskDTO:
        task = await self.repo.get(task_id)
        if not task:
            raise BizException(ErrorCode.BIZ_TASK_NOT_FOUND)
        await self._check_read_access(task, user_id)
        return TaskDTO.model_validate(task, from_attributes=True)

    async def create_task(self, payload: TaskCreate, user_id: int) -> TaskDTO:
        async with self.session.begin():
            task = await self.repo.create(payload, creator_id=user_id)
            await self._record_history(task.id, user_id, "create", None, payload.model_dump())
            return TaskDTO.model_validate(task, from_attributes=True)

    async def update_task(self, task_id: int, payload: TaskUpdate, user_id: int) -> TaskDTO:
        async with self.session.begin():
            task = await self.repo.get(task_id)
            if not task:
                raise BizException(ErrorCode.BIZ_TASK_NOT_FOUND)
            await self._check_write_access(task, user_id)
            old = TaskDTO.model_validate(task, from_attributes=True).model_dump()
            updated = await self.repo.update(task, payload)
            await self._record_history(task_id, user_id, "update", old, payload.model_dump(exclude_none=True))
            return TaskDTO.model_validate(updated, from_attributes=True)

    async def batch_update_status(self, req: BatchUpdateStatusRequest, user_id: int) -> dict:
        # 高危：> 10 项必须经过二次确认
        if len(req.ids) > 10:
            if not req.action_id:
                raise BizException(ErrorCode.BIZ_ACTION_REQUIRED)
            await self.action_svc.verify_action(req.action_id, user_id, "batch_update_task_status")

        async with self.session.begin():
            updated = await self.repo.batch_update_status(req.ids, req.status, viewer_id=user_id)
            return {"updated": updated}

    async def _check_read_access(self, task, user_id: int):
        # FDE 可见: 自己创建的 / 自己负责的 / 所在项目的
        if task.creator_id == user_id or task.assignee_id == user_id:
            return
        if task.project_id and await self.repo.is_project_member(task.project_id, user_id):
            return
        raise BizException(ErrorCode.BIZ_NO_TASK_PERMISSION)

    async def _check_write_access(self, task, user_id: int):
        # 仅创建者 / 责任人 / 项目 owner 可写
        if task.creator_id == user_id or task.assignee_id == user_id:
            return
        if task.project_id and await self.repo.is_project_owner(task.project_id, user_id):
            return
        raise BizException(ErrorCode.BIZ_NO_TASK_PERMISSION)

    async def _record_history(self, task_id, user_id, op, before, after):
        await self.repo.add_history(task_id, user_id, op, before, after)
```

### 5.1 BaseRepository

```python
from typing import Generic, TypeVar, Type
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from app.models.base import Base

T = TypeVar("T", bound=Base)

class BaseRepository(Generic[T]):
    model: Type[T]

    def __init__(self, session: AsyncSession):
        self.session = session

    async def get(self, id: int) -> T | None:
        return await self.session.get(self.model, id)

    async def list_by_ids(self, ids: list[int]) -> list[T]:
        result = await self.session.execute(select(self.model).where(self.model.id.in_(ids)))
        return list(result.scalars().all())

    async def create(self, **kwargs) -> T:
        instance = self.model(**kwargs)
        self.session.add(instance)
        await self.session.flush()
        return instance

    async def soft_delete(self, id: int) -> bool:
        instance = await self.get(id)
        if not instance:
            return False
        instance.is_deleted = 1
        await self.session.flush()
        return True
```

### 5.2 TaskRepository 示例

```python
from sqlalchemy import select, func, and_, update
from app.models.task import Task, TaskHistory
from app.repositories.base import BaseRepository
from app.schemas.task import TaskQuery, TaskCreate, TaskUpdate, TaskStatus

class TaskRepository(BaseRepository[Task]):
    model = Task

    async def search(self, query: TaskQuery, viewer_id: int) -> tuple[list[Task], int]:
        stmt = select(Task).where(Task.is_deleted == 0)
        # 权限过滤
        stmt = stmt.where(or_(Task.creator_id == viewer_id, Task.assignee_id == viewer_id))
        if query.keyword:
            stmt = stmt.where(Task.title.contains(query.keyword))
        if query.status:
            stmt = stmt.where(Task.status.in_(query.status))
        if query.project_id:
            stmt = stmt.where(Task.project_id == query.project_id)

        # 总数
        total = await self.session.scalar(select(func.count()).select_from(stmt.subquery())) or 0
        # 分页
        stmt = stmt.order_by(Task.gmt_create.desc()).offset((query.page - 1) * query.size).limit(query.size)
        result = await self.session.execute(stmt)
        return list(result.scalars().all()), total

    async def create(self, payload: TaskCreate, creator_id: int) -> Task:
        task = Task(**payload.model_dump(), creator_id=creator_id)
        self.session.add(task)
        await self.session.flush()
        return task

    async def update(self, task: Task, payload: TaskUpdate) -> Task:
        for k, v in payload.model_dump(exclude_none=True).items():
            setattr(task, k, v)
        await self.session.flush()
        return task

    async def batch_update_status(self, ids: list[int], status: TaskStatus, viewer_id: int) -> int:
        stmt = (update(Task)
                .where(Task.id.in_(ids), or_(Task.creator_id == viewer_id, Task.assignee_id == viewer_id))
                .values(status=status))
        result = await self.session.execute(stmt)
        return result.rowcount or 0

    async def add_history(self, task_id, user_id, op, before, after):
        self.session.add(TaskHistory(task_id=task_id, user_id=user_id, op=op, before=before, after=after))

    async def is_project_member(self, project_id: int, user_id: int) -> bool: ...
    async def is_project_owner(self, project_id: int, user_id: int) -> bool: ...
```

```python
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request

trace_id_ctx: ContextVar[str] = ContextVar("trace_id", default="")

class TraceMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        trace_id = request.headers.get("X-Request-Id") or str(uuid.uuid4())
        token = trace_id_ctx.set(trace_id)
        try:
            response = await call_next(request)
            response.headers["X-Trace-Id"] = trace_id
            return response
        finally:
            trace_id_ctx.reset(token)
```

### 8.1 路由层（routers/copilot.py）

```python
import json
from fastapi import APIRouter, Depends
from fastapi.responses import StreamingResponse
from app.deps.auth import current_user, UserContext
from app.schemas.copilot import ChatRequest
from app.services.copilot_service import CopilotService

async def chat(req: ChatRequest,
               user: UserContext = Depends(current_user),
               svc: CopilotService = Depends()):
    async def event_stream():
        try:
            async for chunk in svc.chat_stream(req, user):
                yield f"data: {json.dumps(chunk, ensure_ascii=False)}\n\n"
            yield "data: [DONE]\n\n"
        except Exception as e:

    return StreamingResponse(
        event_stream(),
        media_type="text/event-stream",
```

```python
import httpx
from typing import AsyncIterator
from app.config.settings import settings

class CopilotService:
    async def chat_stream(self, req, user) -> AsyncIterator[dict]:
        # 持久化 user msg 到 ai_message
        await self._save_user_message(req, user)

        # 转发到 ai-orchestrator
        async with httpx.AsyncClient(timeout=settings.ai_orchestrator_timeout) as client:
            async with client.stream("POST",
                f"{settings.ai_orchestrator_url}/ai/chat",
                json={**req.model_dump(by_alias=True), "userId": user.id},
            ) as upstream:
                full_response = []
                async for line in upstream.aiter_lines():
                    if not line.startswith("data: "): continue
                    data = line[6:]
                    if data == "[DONE]": break
                    chunk = json.loads(data)
                    if chunk.get("type") == "token": full_response.append(chunk["delta"])
                    yield chunk

        # 持久化 assistant msg
        await self._save_assistant_message(req, user, "".join(full_response))
```

## 九、二次确认机制完整代码（ActionService）

### 9.1 核心代码

```python
import json
import uuid
from datetime import datetime, timedelta
from redis.asyncio import Redis
from app.exceptions.biz import BizException
from app.exceptions.codes import ErrorCode

class ActionService:
    PREFIX = "action:"
    TTL_SECONDS = 60

    def __init__(self, redis: Redis):
        self.redis = redis

    async def stage(self, *, tool_name: str, args: dict, user_id: int,
                    title: str, severity: str, preview: dict,
                    affected_items: list | None = None) -> dict:
        action_id = str(uuid.uuid4())
        payload = {
            "tool_name": tool_name,
            "args": args,
            "user_id": user_id,
            "title": title,
            "severity": severity,
            "preview": preview,
            "affected_items": affected_items or [],
            "status": "pending",
            "created_at": datetime.utcnow().isoformat(),
        }
        await self.redis.setex(f"{self.PREFIX}{action_id}", self.TTL_SECONDS, json.dumps(payload))
        return {
            "action_id": action_id,
            "title": title,
            "severity": severity,
            "preview": preview,
            "affected_items": affected_items,
            "expires_at": (datetime.utcnow() + timedelta(seconds=self.TTL_SECONDS)).isoformat(),
        }

    async def verify_action(self, action_id: str, user_id: int, expected_tool: str) -> dict:
        raw = await self.redis.get(f"{self.PREFIX}{action_id}")
        if not raw:
            raise BizException(ErrorCode.BIZ_ACTION_EXPIRED)
        payload = json.loads(raw)
        if payload["user_id"] != user_id:
            raise BizException(ErrorCode.BIZ_ACTION_USER_MISMATCH)
        if payload["tool_name"] != expected_tool:
            raise BizException(ErrorCode.BIZ_ACTION_TOOL_MISMATCH)
        return payload

    async def execute(self, action_id: str, user_id: int) -> dict:
        payload = await self.verify_action(action_id, user_id, expected_tool=payload_get_tool(action_id))
        await self.redis.delete(f"{self.PREFIX}{action_id}")
        # 派发到具体的工具执行器（dispatcher）
        return await self._dispatch(payload)
```

## 十四、TenantMiddleware 落地补充

> §6.2 仅给出占位，本节补充实际实现（`workspace/api/app/middleware/tenant.py`）。

```python
import structlog
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request

logger = structlog.get_logger()

class TenantMiddleware(BaseHTTPMiddleware):
    """从 X-Tenant-ID Header 提取租户 ID 并注入 structlog 上下文。

    后续阶段会扩展为：
    - 从 JWT Claims 读取 tenant_id（与 Header 双重校验）
    - 将 tenant_id 注入 SQLAlchemy session.info，仓库层自动加 WHERE tenant_id = ?
    """

    async def dispatch(self, request: Request, call_next):
        tenant_id = request.headers.get("X-Tenant-ID", "default")
        structlog.contextvars.bind_contextvars(tenant_id=tenant_id)
        try:
            response = await call_next(request)
            return response
        finally:
            structlog.contextvars.unbind_contextvars("tenant_id")
```
