### 2.3 路由守卫（router/guards.ts，已落地）

```ts
import type { Router } from 'vue-router'
import { useAuthStore } from '@stores/auth'

export function setupGuards(router: Router) {
  router.beforeEach((to, _from, next) => {
    const auth = useAuthStore()
    const requiresAuth = to.meta.requiresAuth !== false

    if (requiresAuth && !auth.isLoggedIn) {
      return next({ path: '/login', query: { redirect: to.fullPath } })
    }
    if (to.path === '/login' && auth.isLoggedIn) {
      return next('/dashboard')
    }

    const title = to.meta.title as string
    document.title = title ? `${title} · FDE 工作台` : 'FDE 工作台'

    next()
  })
}
```

> [!NOTE]
> - `meta.copilot` 字段 = 该页面默认激活的助手 ID（workspace/tasks/project/coach/files/chat），由 CopilotPanel 监听路由变化自动切换。
> - 鉴权读取的是 `useAuthStore().isLoggedIn`（基于 pinia + persisted token），而非直接读 `localStorage`，便于测试 mock。
> - 登录成功后若带 `redirect` query 自动回跳；已登录访问 `/login` 自动重定向到 `/dashboard`。

### 3.2 `<CopilotPanel>`：4 助手统一壳

**Props**：
| 名称 | 类型 | 必填 | 说明 |
|---|---|---|---|
| assistantKey | `'workspace' \| 'tasks' \| 'project' \| 'coach' \| 'files'` | ✓ | 当前激活的助手 |
| context | `Record<string, unknown>` | ✗ | 页面注入上下文（如 projectId） |

**Slots**：无（内部固定结构）

**Events**：
| 名称 | Payload | 说明 |
|---|---|---|
| close | - | 关闭面板 |
| action-confirmed | `{actionId, result}` | 写操作执行成功 |

**结构**：

```vue
<script setup lang="ts">
import { computed, watch } from 'vue'
import { useCopilot } from '@composables/useCopilot'
import MessageList from './MessageList.vue'
import ChatInput from './ChatInput.vue'

const props = defineProps<{
  assistantKey: 'workspace' | 'tasks' | 'project' | 'coach' | 'files'
  context?: Record<string, unknown>
}>()

const emit = defineEmits<{ close: [], 'action-confirmed': [{ actionId: string, result: unknown }] }>()

const { messages, sending, send, stop, activeAssistant, switchAssistant } = useCopilot()
const placeholder = computed(() => activeAssistant.value?.inputPlaceholder ?? '请输入...')

watch(() => props.assistantKey, (k) => switchAssistant(k, props.context), { immediate: true })
</script>

<template>
  <aside class="copilot-panel">
    <header class="copilot-header">
      <span class="copilot-title">{{ activeAssistant?.title }}</span>
      <a-button type="text" size="small" @click="emit('close')">×</a-button>
    </header>
    <MessageList :messages="messages" @action-confirmed="(p) => emit('action-confirmed', p)" />
    <ChatInput :placeholder="placeholder" :sending="sending" @send="send" @stop="stop" />
  </aside>
</template>
```

### 3.4 `<ActionCard>`：写操作预览卡（核心）

```vue
<script setup lang="ts">
import { ref } from 'vue'
import { copilotApi } from '@apis/modules/copilot'
import type { ActionCardData } from '@types/copilot'

const props = defineProps<{ data: ActionCardData }>()
const emit = defineEmits<{ confirmed: [{ actionId: string, result: unknown }] }>()

const loading = ref(false)
const status = ref<'pending' | 'confirmed' | 'cancelled' | 'expired'>('pending')

async function confirm() {
  loading.value = true
  try {
    const result = await copilotApi.executeAction({ actionId: props.data.actionId })
    status.value = 'confirmed'
    emit('confirmed', { actionId: props.data.actionId, result })
  } catch (e: any) {
    if (e.code === 'BIZ_ACTION_EXPIRED') status.value = 'expired'
    throw e
  } finally {
    loading.value = false
  }
}

function cancel() { status.value = 'cancelled' }
</script>

<template>
  <div class="action-card" :class="`action-card--${status}`">
    <div class="action-card__header">
      <span class="badge badge--{{ data.severity }}">{{ data.severity === 'high' ? '高危操作' : '操作预览' }}</span>
      <span class="action-card__title">{{ data.title }}</span>
    </div>
    <div class="action-card__body">
      <pre class="action-card__diff">{{ JSON.stringify(data.preview, null, 2) }}</pre>
      <div v-if="data.affectedItems?.length" class="action-card__affected">
        将影响 {{ data.affectedItems.length }} 项：
        <ul><li v-for="i in data.affectedItems" :key="i.id">{{ i.label }}</li></ul>
      </div>
    </div>
    <footer class="action-card__footer">
      <template v-if="status === 'pending'">
        <a-button @click="cancel">取消</a-button>
        <a-button type="primary" :loading="loading" @click="confirm">确认执行</a-button>
      </template>
      <span v-else-if="status === 'confirmed'" class="text-success">✓ 已执行</span>
      <span v-else-if="status === 'cancelled'" class="text-muted">已取消</span>
      <span v-else-if="status === 'expired'" class="text-danger">⚠ 已过期，请重新发起</span>
    </footer>
  </div>
</template>
```

### 6.1 `apis/http.ts`（Axios 实例 + 拦截器）

```ts
import axios, { type AxiosInstance, type InternalAxiosRequestConfig } from 'axios'
import { message } from 'ant-design-vue'
import { useAuthStore } from '@stores/auth'
import router from '@/router'

export const http: AxiosInstance = axios.create({
  baseURL: '/api/v1',
  timeout: 15000,
  headers: { 'Content-Type': 'application/json' },
})

let isRefreshing = false
let pendingQueue: Array<(t: string) => void> = []

export function setupAxiosInterceptors() {
  http.interceptors.request.use((config: InternalAxiosRequestConfig) => {
    const token = localStorage.getItem('access_token')
    if (token) config.headers.Authorization = `Bearer ${token}`
    config.headers['X-Request-Id'] = crypto.randomUUID()
    return config
  })

  http.interceptors.response.use(
    (resp) => {
      const { code, data, message: msg } = resp.data
      if (code !== 0 && code !== 'SUCCESS') {
        return Promise.reject({ code, message: msg })
      }
      return data
    },
    async (err) => {
      const status = err.response?.status
      if (status === 401) {
        const auth = useAuthStore()
        if (!auth.refreshToken) { router.push('/login'); return Promise.reject(err) }
        if (isRefreshing) {
          return new Promise((resolve) => pendingQueue.push((t) => {
            err.config.headers.Authorization = `Bearer ${t}`
            resolve(http(err.config))
          }))
        }
        isRefreshing = true
        try {
          await auth.refresh()
          pendingQueue.forEach((cb) => cb(auth.accessToken))
          pendingQueue = []
          err.config.headers.Authorization = `Bearer ${auth.accessToken}`
          return http(err.config)
        } catch (e) {
          auth.logout(); router.push('/login'); return Promise.reject(e)
        } finally { isRefreshing = false }
      }
      message.error(err.response?.data?.message || err.message || '网络错误')
      return Promise.reject(err)
    },
  )
}
```

### 6.2 `apis/sse.ts`（SSE 客户端，与实际落地一致）

> 浏览器原生 `EventSource` 不支持 POST + body，故使用 `fetch + ReadableStream` 自实现，避免引入 `@microsoft/fetch-event-source` 第三方依赖。返回 `AbortController`，调用方持有以中断流。

```ts
interface OpenSseOptions {
  url: string                            // 相对路径，内部自动拼 /api/v1
  body?: Record<string, unknown>
  onMessage: (data: unknown) => void     // 收到 data: 行的回调；JSON.parse 失败时回传原文
  onDone?: () => void                    // 收到 [DONE] 或流自然结束
  onError?: (error: Error) => void       // 非 AbortError 才回调
}

export function openSse(options: OpenSseOptions): AbortController {
  const { url, body, onMessage, onDone, onError } = options
  const controller = new AbortController()
  const token = localStorage.getItem('access_token')

  fetch(`/api/v1${url}`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    },
    body: body ? JSON.stringify(body) : undefined,
    signal: controller.signal,
  })
    .then(async (response) => {
      if (!response.ok || !response.body) throw new Error(`SSE ${response.status}`)
      const reader = response.body.getReader()
      const decoder = new TextDecoder()
      let buffer = ''
      while (true) {
        const { done, value } = await reader.read()
        if (done) break
        buffer += decoder.decode(value, { stream: true })
        const lines = buffer.split('\n')
        buffer = lines.pop() || ''
        for (const line of lines) {
          const trimmed = line.trim()
          if (!trimmed.startsWith('data: ')) continue
          const data = trimmed.slice(6)
          if (data === '[DONE]') { onDone?.(); return }
          try { onMessage(JSON.parse(data)) } catch { onMessage(data) }
        }
      }
      onDone?.()
    })
    .catch((err: Error) => { if (err.name !== 'AbortError') onError?.(err) })

  return controller
}
```

> **关键约束**：
> - **认证用 Header（Bearer Token），不放 URL query**，避免 token 落入服务端访问日志。
> - **`AbortError` 不向上抛**：用户主动 abort 视为正常关闭，避免 toast 报错。
> - **JSON 解析降级**：解析失败时回传原始字符串，由消费方决定如何处理（兜底兼容）。

```ts
copilotApi.chat(
  {
    assistantId: assistantIdMap[assistantType] || 'chat',
    message: content,
    sessionId: getSessionId(assistantType),
    mode: 'smart',
    context: { assistantType },
  },
  (chunk) => {                                  // onMessage
```

### 7.2 `useMention.ts`：@ 引用搜索（与实际一致）

```ts
import { ref, watch } from 'vue'
import { debounce } from 'lodash-es'
import { mentionsApi } from '@apis/modules/mentions'
import type { MentionItem, MentionType } from '@types/copilot'

export function useMention() {
  const query = ref('')
  const type = ref<MentionType>('task')
  const items = ref<MentionItem[]>([])
  const loading = ref(false)

  const fetch = debounce(async () => {
    if (!query.value) { items.value = []; return }
    loading.value = true
    try { items.value = await mentionsApi.search({ q: query.value, type: type.value }) }
    finally { loading.value = false }
  }, 300)

  watch([query, type], fetch)
  return { query, type, items, loading }
}
```
