### 3.2 `<CopilotPanel>`：4 助手统一壳

**Props**：
| 名称 | 类型 | 必填 | 说明 |
|---|---|---|---|
| assistantKey | `'workspace' \| 'tasks' \| 'project' \| 'coach' \| 'files'` | ✓ | 当前激活的助手 |
| context | `Record<string, unknown>` | ✗ | 页面注入上下文（如 projectId） |

**Slots**：无（内部固定结构）

**Events**：
| 名称 | Payload | 说明 |
|---|---|---|
| close | - | 关闭面板 |
| action-confirmed | `{actionId, result}` | 写操作执行成功 |

**结构**：

```vue
<script setup lang="ts">
import { computed, watch } from 'vue'
import { useCopilot } from '@composables/useCopilot'
import MessageList from './MessageList.vue'
import ChatInput from './ChatInput.vue'

const props = defineProps<{
  assistantKey: 'workspace' | 'tasks' | 'project' | 'coach' | 'files'
  context?: Record<string, unknown>
}>()

const emit = defineEmits<{ close: [], 'action-confirmed': [{ actionId: string, result: unknown }] }>()

const { messages, sending, send, stop, activeAssistant, switchAssistant } = useCopilot()
const placeholder = computed(() => activeAssistant.value?.inputPlaceholder ?? '请输入...')

watch(() => props.assistantKey, (k) => switchAssistant(k, props.context), { immediate: true })
</script>

<template>
  <aside class="copilot-panel">
    <header class="copilot-header">
      <span class="copilot-title">{{ activeAssistant?.title }}</span>
      <a-button type="text" size="small" @click="emit('close')">×</a-button>
    </header>
    <MessageList :messages="messages" @action-confirmed="(p) => emit('action-confirmed', p)" />
    <ChatInput :placeholder="placeholder" :sending="sending" @send="send" @stop="stop" />
  </aside>
</template>
```

### 3.4 `<ActionCard>`：写操作预览卡（核心）

```vue
<script setup lang="ts">
import { ref } from 'vue'
import { copilotApi } from '@apis/modules/copilot'
import type { ActionCardData } from '@types/copilot'

const props = defineProps<{ data: ActionCardData }>()
const emit = defineEmits<{ confirmed: [{ actionId: string, result: unknown }] }>()

const loading = ref(false)
const status = ref<'pending' | 'confirmed' | 'cancelled' | 'expired'>('pending')

async function confirm() {
  loading.value = true
  try {
    const result = await copilotApi.executeAction({ actionId: props.data.actionId })
    status.value = 'confirmed'
    emit('confirmed', { actionId: props.data.actionId, result })
  } catch (e: any) {
    if (e.code === 'BIZ_ACTION_EXPIRED') status.value = 'expired'
    throw e
  } finally {
    loading.value = false
  }
}

function cancel() { status.value = 'cancelled' }
</script>

<template>
  <div class="action-card" :class="`action-card--${status}`">
    <div class="action-card__header">
      <span class="badge badge--{{ data.severity }}">{{ data.severity === 'high' ? '高危操作' : '操作预览' }}</span>
      <span class="action-card__title">{{ data.title }}</span>
    </div>
    <div class="action-card__body">
      <pre class="action-card__diff">{{ JSON.stringify(data.preview, null, 2) }}</pre>
      <div v-if="data.affectedItems?.length" class="action-card__affected">
        将影响 {{ data.affectedItems.length }} 项：
        <ul><li v-for="i in data.affectedItems" :key="i.id">{{ i.label }}</li></ul>
      </div>
    </div>
    <footer class="action-card__footer">
      <template v-if="status === 'pending'">
        <a-button @click="cancel">取消</a-button>
        <a-button type="primary" :loading="loading" @click="confirm">确认执行</a-button>
      </template>
      <span v-else-if="status === 'confirmed'" class="text-success">✓ 已执行</span>
      <span v-else-if="status === 'cancelled'" class="text-muted">已取消</span>
      <span v-else-if="status === 'expired'" class="text-danger">⚠ 已过期，请重新发起</span>
    </footer>
  </div>
</template>
```

export const useAuthStore = defineStore('auth', {

      this.refreshToken = ''
      localStorage.removeItem('access_token')
      localStorage.removeItem('refresh_token')

import { defineStore } from 'pinia'
import type { AssistantKey, CopilotMessage } from '@types/copilot'

interface SessionState {
  messages: CopilotMessage[]
  sessionId: string | null
  context: Record<string, unknown>
}

export const useCopilotStore = defineStore('copilot', {
  state: () => ({
    activeKey: 'workspace' as AssistantKey,
    sessions: {} as Record<AssistantKey, SessionState>,
    sending: false,
  }),
  getters: {
    currentSession: (s) => s.sessions[s.activeKey] ?? { messages: [], sessionId: null, context: {} },
  },
  actions: {
    setActive(key: AssistantKey, context: Record<string, unknown> = {}) {
      this.activeKey = key
      if (!this.sessions[key]) this.sessions[key] = { messages: [], sessionId: null, context }
      else this.sessions[key].context = { ...this.sessions[key].context, ...context }
    },
    appendMessage(msg: CopilotMessage) { this.currentSession.messages.push(msg) },
    appendAssistantChunk(id: string, chunk: string) {
      const last = this.currentSession.messages.find((m) => m.id === id)
      if (last) last.content += chunk
    },
    reset(key: AssistantKey) { this.sessions[key] = { messages: [], sessionId: null, context: {} } },
  },
})

### 6.1 `apis/http.ts`（Axios 实例 + 拦截器）

```ts
import axios, { type AxiosInstance, type InternalAxiosRequestConfig } from 'axios'
import { message } from 'ant-design-vue'
import { useAuthStore } from '@stores/auth'
import router from '@/router'

export const http: AxiosInstance = axios.create({
  baseURL: '/api/v1',
  timeout: 15000,
  headers: { 'Content-Type': 'application/json' },
})

let isRefreshing = false
let pendingQueue: Array<(t: string) => void> = []

export function setupAxiosInterceptors() {
  http.interceptors.request.use((config: InternalAxiosRequestConfig) => {
    const token = localStorage.getItem('access_token')
    if (token) config.headers.Authorization = `Bearer ${token}`
    config.headers['X-Request-Id'] = crypto.randomUUID()
    return config
  })

  http.interceptors.response.use(
    (resp) => {
      const { code, data, message: msg } = resp.data
      if (code !== 0 && code !== 'SUCCESS') {
        return Promise.reject({ code, message: msg })
      }
      return data
    },
    async (err) => {
      const status = err.response?.status
      if (status === 401) {
        const auth = useAuthStore()
        if (!auth.refreshToken) { router.push('/login'); return Promise.reject(err) }
        if (isRefreshing) {
          return new Promise((resolve) => pendingQueue.push((t) => {
            err.config.headers.Authorization = `Bearer ${t}`
            resolve(http(err.config))
          }))
        }
        isRefreshing = true
        try {
          await auth.refresh()
          pendingQueue.forEach((cb) => cb(auth.accessToken))
          pendingQueue = []
          err.config.headers.Authorization = `Bearer ${auth.accessToken}`
          return http(err.config)
        } catch (e) {
          auth.logout(); router.push('/login'); return Promise.reject(e)
        } finally { isRefreshing = false }
      }
      message.error(err.response?.data?.message || err.message || '网络错误')
      return Promise.reject(err)
    },
  )
}
```

### 6.2 `apis/sse.ts`（SSE 客户端）

```ts
import { fetchEventSource, type EventSourceMessage } from '@microsoft/fetch-event-source'

export interface SseOptions {
  url: string
  body: unknown
  signal?: AbortSignal
  onMessage: (data: any) => void
  onDone?: () => void
  onError?: (err: unknown) => void
}

export async function openSse(opts: SseOptions): Promise<void> {
  const token = localStorage.getItem('access_token')
  await fetchEventSource(opts.url, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: token ? `Bearer ${token}` : '',
    },
    body: JSON.stringify(opts.body),
    signal: opts.signal,
    onmessage(ev: EventSourceMessage) {
      if (ev.data === '[DONE]') { opts.onDone?.(); return }
      try { opts.onMessage(JSON.parse(ev.data)) }
      catch (e) { opts.onError?.(e) }
    },
    onerror(err) { opts.onError?.(err); throw err /* 不自动重连 */ },
    openWhenHidden: true,
  })
}
```

## 七、Composables 设计

### 7.1 `useSSEChat.ts`：SSE 流式对话

```ts
import { ref, onBeforeUnmount } from 'vue'
import { openSse } from '@apis/sse'

export function useSSEChat() {
  const sending = ref(false)
  const controller = ref<AbortController | null>(null)

  async function send(opts: { url: string; body: unknown; onChunk: (chunk: any) => void; onDone?: () => void }) {
    sending.value = true
    controller.value = new AbortController()
    try {
      await openSse({
        url: opts.url,
        body: opts.body,
        signal: controller.value.signal,
        onMessage: opts.onChunk,
        onDone: opts.onDone,
      })
    } finally { sending.value = false }
  }

  function stop() { controller.value?.abort() }
  onBeforeUnmount(stop)
  return { sending, send, stop }
}
```

### 7.2 `useCopilot.ts`：4 助手统一接入

```ts
import { computed } from 'vue'
import { useCopilotStore } from '@stores/copilot'
import { useSSEChat } from './useSSEChat'
import { ASSISTANTS } from '@/config/assistants'
import type { AssistantKey } from '@types/copilot'

export function useCopilot() {
  const store = useCopilotStore()
  const { sending, send: sseSend, stop } = useSSEChat()

  const activeAssistant = computed(() => ASSISTANTS[store.activeKey])
  const messages = computed(() => store.currentSession.messages)

  function switchAssistant(key: AssistantKey, context: Record<string, unknown> = {}) { store.setActive(key, context) }

  async function send(text: string) {
    const userMsgId = crypto.randomUUID()
    const aiMsgId = crypto.randomUUID()
    store.appendMessage({ id: userMsgId, role: 'user', content: text, gmtCreate: new Date().toISOString() })
    store.appendMessage({ id: aiMsgId, role: 'assistant', content: '', gmtCreate: new Date().toISOString() })

    await sseSend({
      url: '/api/v1/copilot/chat',
      body: {
        assistantId: store.activeKey,
        sessionId: store.currentSession.sessionId,
        message: text,
        context: store.currentSession.context,
      },
      onChunk: (chunk) => {
        if (chunk.type === 'token') store.appendAssistantChunk(aiMsgId, chunk.delta)
        else if (chunk.type === 'card') {
          const m = store.currentSession.messages.find((m) => m.id === aiMsgId)
          if (m) m.card = chunk.card
        } else if (chunk.type === 'session_id') {
          store.currentSession.sessionId = chunk.sessionId
        }
      },
    })
  }

  return { activeAssistant, messages, sending, send, stop, switchAssistant }
}
```

### 7.3 `useMention.ts`：@ 引用搜索

```ts
import { ref, watch } from 'vue'
import { debounce } from 'lodash-es'
import { mentionsApi } from '@apis/modules/mentions'
import type { MentionItem, MentionType } from '@types/copilot'

export function useMention() {
  const query = ref('')
  const type = ref<MentionType>('task')
  const items = ref<MentionItem[]>([])
  const loading = ref(false)

  const fetch = debounce(async () => {
    if (!query.value) { items.value = []; return }
    loading.value = true
    try { items.value = await mentionsApi.search({ q: query.value, type: type.value }) }
    finally { loading.value = false }
  }, 300)

  watch([query, type], fetch)
  return { query, type, items, loading }
}
```
